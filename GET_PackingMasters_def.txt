CREATE FUNCTION [dbo].[GET_PackingMasters]
                                                                                                                                                                                                                   
(	
                                                                                                                                                                                                                                                           
		@PCAID				VARCHAR(24)		=	'',
                                                                                                                                                                                                                               
		@COMPANYID			VARCHAR(24)		=	'',
                                                                                                                                                                                                                            
		@DEPTID				VARCHAR(24)		=	''
                                                                                                                                                                                                                               
)
                                                                                                                                                                                                                                                            
RETURNS TABLE 
                                                                                                                                                                                                                                               
AS
                                                                                                                                                                                                                                                           
RETURN 
                                                                                                                                                                                                                                                      
(
                                                                                                                                                                                                                                                            
		SELECT		COALESCE(PCAID					,	'')		AS	PCAID				
                                                                                                                                                                                                             
					,COALESCE(PAID					,	'')		AS	PAID				
                                                                                                                                                                                                                   
					,COALESCE(PCTITLE				,	'')		AS	PCTITLE			
                                                                                                                                                                                                               
					,COALESCE(PCSERIAL				,	'')		AS	PCSERIAL			
                                                                                                                                                                                                             
					,COALESCE(PCQUANTITYUOM			,	'')		AS	PCQUANTITYUOM		
                                                                                                                                                                                                     
					,COALESCE(PCQUANTITYAMOUNT		,	'')		AS	PCQUANTITYAMOUNT
                                                                                                                                                                                                  
	
                                                                                                                                                                                                                                                            

                                                                                                                                                                                                                                                             
		FROM		tblPackingMasters
                                                                                                                                                                                                                                    
		WHERE		ISDELETED = 0
                                                                                                                                                                                                                                       
			AND		COMPANYID	=	@COMPANYID
                                                                                                                                                                                                                               
			AND		DEPTID		=	@DEPTID		
                                                                                                                                                                                                                                  
			AND		PCAID	LIKE	@PCAID
                                                                                                                                                                                                                                    
)
