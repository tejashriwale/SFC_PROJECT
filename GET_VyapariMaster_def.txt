CREATE FUNCTION [dbo].[GET_VyapariMaster]
                                                                                                                                                                                                                    
(	
                                                                                                                                                                                                                                                           
		@VPAID				VARCHAR(24)		=	'%',
                                                                                                                                                                                                                              
		@KEYWORD			VARCHAR(MAX)	=	'%',
                                                                                                                                                                                                                             
		@COMPANYID			VARCHAR(24)		=	'',
                                                                                                                                                                                                                            
		@DEPTID				VARCHAR(24)		=	''
                                                                                                                                                                                                                               
		
                                                                                                                                                                                                                                                           
		
                                                                                                                                                                                                                                                           
)
                                                                                                                                                                                                                                                            
RETURNS TABLE 
                                                                                                                                                                                                                                               
AS
                                                                                                                                                                                                                                                           
RETURN 
                                                                                                                                                                                                                                                      
(
                                                                                                                                                                                                                                                            
	
                                                                                                                                                                                                                                                            
		SELECT 
                                                                                                                                                                                                                                                    
			   
                                                                                                                                                                                                                                                       

                                                                                                                                                                                                                                                             
				COALESCE(VPAID				,	'')		AS							VPAID
                                                                                                                                                                                                                  
			   ,COALESCE(CONVERT(VARCHAR(16), VREGISTRATION,	106)	,	'')	AS	VREGISTRATION
                                                                                                                                                                              
			   ,COALESCE(VNAME				,	'')		AS							VNAME	
                                                                                                                                                                                                              
			   ,COALESCE(OTP				,	'')		AS							OTP	
                                                                                                                                                                                                                  
			   ,COALESCE(VPPHOTO			,	'')		AS						    VPPHOTO	
                                                                                                                                                                                                        
			   ,COALESCE(VPAADHAR			,	'')		AS							VPAADHAR	
                                                                                                                                                                                                         
			   ,COALESCE(VPPAN				,	'')		AS							VPPAN	
                                                                                                                                                                                                              
			   ,COALESCE(VSHORTNAME			,	'')		AS							VSHORTNAME	
                                                                                                                                                                                                     
			   ,COALESCE(VTYPE				,	'')		AS							VTYPE				
                                                                                                                                                                                                           
			   ,COALESCE(VMOBLIE			,	'')		AS							VMOBLIE			
                                                                                                                                                                                                         
			   ,COALESCE(VEMAIL				,	'')		AS							VEMAIL				
                                                                                                                                                                                                         
			   ,COALESCE(VAADHAR			,	'')		AS							VAADHAR			
                                                                                                                                                                                                         
			   ,COALESCE(VPAN				,	'')		AS							VPAN				
                                                                                                                                                                                                             
			   ,COALESCE(CONVERT(VARCHAR(16), VBIRTHDAYDATE,	106)	,	'')	AS	VBIRTHDAYDATE		
                                                                                                                                                                            
			   ,COALESCE(CONVERT(VARCHAR(16), VANIVERSARYDATE,	106)	,	'')	AS	VANIVERSARYDATE	
                                                                                                                                                                         
			   ,COALESCE(VSTATUS			,	'')		AS							VSTATUS			
                                                                                                                                                                                                         
			   ,COALESCE(VADDRESS			,	'')		AS							VADDRESS			
                                                                                                                                                                                                       
			   ,COALESCE(VAREA				,	'')		AS							VAREA				
                                                                                                                                                                                                           
			   ,COALESCE(VPINCODE			,	'')		AS							VPINCODE			
                                                                                                                                                                                                       
			   ,COALESCE(VCITY				,	'')		AS							VCITY				
                                                                                                                                                                                                           
			   ,COALESCE(VSTATE				,	'')		AS							VSTATE				
                                                                                                                                                                                                         
			   ,COALESCE(VSAMEADDRESS		,	'')		AS							VSAMEADDRESS		
                                                                                                                                                                                                 
			   ,COALESCE(VGSTIN				,	'')		AS							VGSTIN				
                                                                                                                                                                                                         
			   ,COALESCE(VCOMPNAME			,	'')		AS							VCOMPNAME			
                                                                                                                                                                                                     
			   ,COALESCE(VCOMPPAN			,	'')		AS							VCOMPPAN			
                                                                                                                                                                                                       
			   ,COALESCE(VCOMPADDRESS		,	'')		AS							VCOMPADDRESS		
                                                                                                                                                                                                 
			   ,COALESCE(VCOMPAREA			,	'')		AS							VCOMPAREA			
                                                                                                                                                                                                     
			   ,COALESCE(VCOMPPINCODE		,	'')		AS							VCOMPPINCODE		
                                                                                                                                                                                                 
			   ,COALESCE(VCOMPCITY			,	'')		AS							VCOMPCITY			
                                                                                                                                                                                                     
			   ,COALESCE(VCOMPSTATE			,	'')		AS							VCOMPSTATE			
                                                                                                                                                                                                   
			   ,COALESCE(VIFSCCODE			,	'')		AS							VIFSCCODE			
                                                                                                                                                                                                     
			   ,COALESCE(VACCOUNTNUMBER		,	'')		AS							VACCOUNTNUMBER		
                                                                                                                                                                                             
			   ,COALESCE(VACCOUNTNAME		,	'')		AS							VACCOUNTNAME		
                                                                                                                                                                                                 
			   ,COALESCE(VBANKNAME			,	'')		AS							VBANKNAME			
                                                                                                                                                                                                     
			   ,COALESCE(VBRANCHNAME		,	'')		AS							VBRANCHNAME		
                                                                                                                                                                                                   
			   ,COALESCE(VPSCHEME		    ,	'')		AS							VPSCHEME
                                                                                                                                                                                                       
			   ,COALESCE(IR.ImplicationTitle         ,   '')     AS					VYAPARITYPE
                                                                                                                                                                                   
			   
                                                                                                                                                                                                                                                       
		FROM	tblVyapariMaster V
                                                                                                                                                                                                                                    

                                                                                                                                                                                                                                                             
		LEFT JOIN ST_Implication IR ON IR.IsDeleted=0 AND V.VTYPE = IR.ImplicationValue AND IR.ImplicationGroup ='VTYPE'
                                                                                                                                           

                                                                                                                                                                                                                                                             
		WHERE	V.ISDELETED	=	0
                                                                                                                                                                                                                                      
			AND	V.COMPANYID	=	@COMPANYID
                                                                                                                                                                                                                              
			AND	V.DEPTID		=	@DEPTID
                                                                                                                                                                                                                                   
			AND	V.VPAID	LIKE	@VPAID
                                                                                                                                                                                                                                   
				AND	(
                                                                                                                                                                                                                                                    
								VNAME			LIKE    '%'		+   @KEYWORD	+	'%'
                                                                                                                                                                                                              
						    OR  VTYPE			LIKE	'%'		+	@KEYWORD	+	'%'				
                                                                                                                                                                                                         
							OR	VCOMPNAME			LIKE	'%'		+	@KEYWORD	+	'%'
                                                                                                                                                                                                             
							
                                                                                                                                                                                                                                                      
							
                                                                                                                                                                                                                                                      
			
                                                                                                                                                                                                                                                          
				)
                                                                                                                                                                                                                                                        

                                                                                                                                                                                                                                                             

                                                                                                                                                                                                                                                             

                                                                                                                                                                                                                                                             
				--select * from GET_VyapariMaster('%','%','COMP123456789','D001') order by CONVERT(date, VREGISTRATION, 106) desc;
                                                                                                                                       
)
